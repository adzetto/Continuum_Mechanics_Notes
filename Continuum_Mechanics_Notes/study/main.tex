  \def\pr@a{#1}\def\pr@b{}%
  \ifx\pr@a\pr@b
    \addcontentsline{toc}{subsection}{Problem \theproblemenv}%
  \else
    \addcontentsline{toc}{subsection}{%
      \texorpdfstring{Problem \theproblemenv: #1}{Problem \theproblemenv}}%
  \fi}

  {\begin{problemenv}[#1]\prob@toc{#1}}%
